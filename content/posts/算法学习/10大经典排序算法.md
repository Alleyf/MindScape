# 1 归并排序

> 1. 快慢指针定位中间节点（mid=slow.next），并切断一分为二
> 2. 分别递归排序两个子链表
> 3. 合并两个有序子链表，返回头结点

```java
import java.util.*;
import java.lang.*;

public class Main{

   public static void main(String[] args){
       Scanner in = new Scanner(System.in);
       Node dummy = new Node(-1);
       Node cur = dummy;
       while(in.hasNextInt()){
           Node node = new Node(in.nextInt());
           cur.next = node;
           cur = cur.next;
       }
       Node head = Solve.mergeSort(dummy.next);
       Node.print(head);
   }
}

class Solve{
   public static Node mergeSort(Node root){
       // 边界判断
       if(root==null||root.next == null){
           return root;
       }
       // 快慢指针定位中间节点
       Node slow = root,fast = root.next;
       while(fast!=null&&fast.next!=null){
           fast = fast.next.next;
           slow = slow.next;
       }
       Node mid = slow.next;
       // 断开原链表链接
       slow.next = null;
       // 递归排序两个子链表
       Node heada = mergeSort(root);
       Node headb = mergeSort(mid);
       // 合并两个有序链表
       Node dummy = new Node(-1);
       Node cur = dummy;
       while(heada!=null&&headb!=null){
           if(heada.val<=headb.val){
               cur.next = heada;
               heada = heada.next;
           }else{
               cur.next = headb;
               headb = headb.next;
           }
           cur = cur.next;
       }
       // 链接未处理完的剩余部分
       cur.next = heada!=null?heada:headb;
       // 返回合并后的链表头结点
       return dummy.next;
   }
}

class Node{
   int val;
   Node next;

   Node(int val){
       this.val = val;
       this.next = null;
   }

   public static void print(Node head){
           while(head!=null){
           if(head.next!=null){
               System.out.print(head.val+" ");
           }else{
               System.out.print(head.val);
           }
           head = head.next;
       }

   }

}
```
